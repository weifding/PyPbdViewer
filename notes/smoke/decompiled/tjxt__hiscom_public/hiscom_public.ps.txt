public function long uf_hextodec (string hexnumber);//将16进制字符转换成10进制数值

long ll_i, ll_Len, ll_HexBit, ll_DecNumber
string ls_HexBit

HexNumber = Trim(HexNumber)
ll_Len = Len(HexNumber)
if ll_Len > 0 then
	HexNumber = Reverse(HexNumber)
	for ll_i = 1 to ll_Len
		ls_HexBit = Mid(HexNumber, ll_i, 1)
		if IsNumber(ls_HexBit) then
			ll_HexBit = Long(ls_HexBit)
		elseif Match(ls_HexBit, "[A-F]") then
			ll_HexBit = Asc(ls_HexBit) - 55
		elseif Match(ls_HexBit, "[a-f]") then
			ll_HexBit = Asc(ls_HexBit) - 87
		else
			return -1
		end if
		ll_DecNumber += (ll_HexBit * 16^(ll_i - 1))
	next
end if

return ll_DecNumber

end function

public function string uf_getline (string strtext, long linenumber);//返回字符串strtext的第linenumber行

constant string crlf = char(13) + char(10)

string ls_data

if IsNull(strText) or IsNull(LineNumber) or LineNumber <= 0 then return ""

	strText += CRLF
end if

ll_pos1 = 1
ll_pos2 = pos(strText, crlf)
ll_count = 0

do while ll_pos2 > 0
	ll_count ++
	if ll_count = LineNumber then
		return Mid(strText, ll_pos1, ll_pos2 - ll_pos1)
	end if

	ll_pos2 = pos(strText, crlf, ll_pos1)
loop

return ""
end function

public function string uf_cutstr (string code, string delim, integer index);long ll_begin, ll_end = 1, ll_count, ll_dlen, ll_oldbegin
string ls_ret

if IsNull(code) or code = '' then return ''
if IsNull(delim) or delim = '' then return ''
if IsNull(index) or index <= 0 then return ''

ll_dlen = Len(delim)
if left(code, ll_dlen) <> delim then code = delim + code
if right(code, ll_dlen) <> delim then code += delim

do until ll_count = index
	ll_begin = Pos(code, delim, ll_end)
	ll_end = Pos(code, delim, ll_begin + 1)
	if ll_end = 0 then return ""
	ll_oldbegin = ll_begin
	ll_count ++
loop
ls_ret = Mid(code, ll_oldbegin + ll_dlen,  ll_end - ll_oldbegin - ll_dlen)

return ls_ret
end function

public function integer uf_putarray (string data, string delim, ref string unit[]);string ls_unit
long ll_count, ll_pos, ll_len

ll_len = Len(delim)
if Right(data, ll_len) <> delim then
	data += delim
end if
ll_pos = Pos(data, delim)
do while ll_pos > 0
	yield()
	ll_count ++
	unit[ll_count] = Left(data, ll_pos - 1)
	data = Mid(data, ll_pos + ll_len)
	ll_pos = Pos(data, delim)
loop

return ll_count


end function

event ue_process(readonly string as_data);string ls_name,ls_temp

in_hiscomsys.uf_dump(is_path+'\raw\'+ls_name+'.txt', as_data)

event ue_port();TRY
	If ole_comm.Object.PortOpen Then	////if opened
		ole_comm.Object.RtsEnable = False
		ole_comm.Object.PortOpen = False			///closed it
		gf_hiscomlog(is_path,"关闭端口 " + string(ole_comm.Object.CommPort))
	
	Else
		ole_comm.Object.Settings = in_hiscomdcb.is_commstr
		ole_comm.Object.CommPort = in_hiscomdcb.ii_port
		ole_comm.object.InBufferSize = in_hiscomdcb.ii_inbuffersize
		ole_comm.Object.HandShaking = in_hiscomdcb.ii_handshaking
		ole_comm.Object.InputMode = in_hiscomdcb.ii_inputmode
		ole_comm.Object.PortOpen = True
		ole_comm.Object.RtsEnable = True
		gf_hiscomlog(is_path,"打开端口COM" + String(Ole_comm.Object.CommPort) +& 				' 参数：' + Ole_comm.Object.Settings + " 二进制流解码=" + string(in_hiscomdcb.ib_usebinary))
	End If

		
CATCH (Throwable  opene)
	if isvalid(in_hiscomdcb) then
		gf_hiscomlog('',"打开端口COM" +string(in_hiscomdcb.ii_port)+" 失败：" + opene.text)
	else
		gf_hiscomlog('',"打开端口COM失败：" + opene.text)
	end if
	if gs_rtxserver='' then gs_rtxserver='rtx.fey.com'
	if isvalid(gn_rtx) then
		gn_rtx.uf_sendmsg(gs_rtxserver,'信息技术科',	'HIS串口通讯',"打开通讯窗口失败：~r~n"+ opene.text) 
	end if
	 
FINALLY
	
END TRY


end event

in_hiscomsys.uf_dump(is_path+'\raw\'+ls_name+'.txt', ab_data)

end event

event open;TRY
	string ls_yiqileixin
	this.visible=false
	
	in_hiscomsys=create n_hiscomsys
	is_path=in_hiscomsys.is_path 
	
	ls_yiqileixin=message.stringparm
	if isnull(ls_yiqileixin) then ls_yiqileixin=''
	if ls_yiqileixin='' then
		gf_hiscomlog(is_path,'系统参数为空')
		close(this)
		return 
	end if
	
	
	if in_hiscomsys.uf_init( ls_yiqileixin)<0 then
	 
		gf_hiscomlog(is_path,'初始化失败')
		close(this)
		return 
	end if

	 
	TRY
		in_hiscomdcb = create using 'n_'+ ls_yiqileixin
		 
	CATCH (Throwable  in_hiscomdcbe)
		gf_hiscomlog(is_path,'创建解码模块失败，可能对象 ：'+  'n_'+ ls_yiqileixin + " 不存在，或者函数调用参数不对！" + in_hiscomdcbe.text)
		close(this)
		return 
	FINALLY 
	END TRY
	
	
	
	IF in_hiscomdcb.uf_init(in_hiscomsys) =false Then 
		gf_hiscomlog(is_path,'解码模块初使化出错')
		close(this)
		Return 
	End IF
	in_hiscomdcb.iw_parentwindow=this

	//判断winsock6.0是否注册
	String ls_value
	ls_value = ''
	RegistryGet("HKEY_CLASSES_ROOT\CLSID\{648A5600-2C6E-101B-82B6-000000000014}\InprocServer32","",RegString!, ls_value)
	IF IsNull(ls_value) Or ls_value = '' THEN
		 gf_hiscomlog(is_path,"打开通讯碊AT* 翱谑О埽捍�口通讯控件未注� !")
	
	END IF	
CATCH (Throwable  opene)
	
	gf_hiscomlog('',"打开通讯窗口失败：" + opene.text)
	if gs_rtxserver='' then gs_rtxserver='rtx.fey.com'
	if isvalid(gn_rtx) then
		gn_rtx.uf_sendmsg(gs_rtxserver,'信息技术科',	'HIS串口通讯',"打开通讯窗口失败：~r~n"+ opene.text) 
	end if
	close(this)
	return
FINALLY
	
END TRY


gf_hiscomlog(is_path,'启动系统')
Event  ue_port()
end event

event close;IF ole_comm.Object.portopen Then 
	Event ue_port()
End If

if isvalid(in_hiscomdcb) then
	destroy  in_hiscomdcb
end if

if isvalid(in_hiscomsys) then
	destroy in_hiscomsys
end if
gf_hiscomlog(is_path,"退出系统")
end event
