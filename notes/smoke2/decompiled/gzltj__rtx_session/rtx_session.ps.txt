loop

return
end subroutine

protected function boolean of_getbit (long al_decimal, unsignedinteger ai_bit);Boolean lb_null

If IsNull(al_decimal) or IsNull(ai_bit) then
    SetNull(lb_null)
    Return lb_null
End If

If Int(Mod(al_decimal / (2 ^(ai_bit - 1)), 2)) > 0 Then
    Return True

protected function long of_bitwiseand (long al_value1, long al_value2);Integer   li_Cnt
Long    ll_Result
Boolean   lb_Value1[32], lb_Value2[32]


If IsNull(al_Value1) Or IsNull(al_Value2) Then
	SetNull(ll_Result)
	Return ll_Result
End If

For li_Cnt = 1 To 32
	lb_Value1[li_Cnt] = of_getbit(al_Value1, li_Cnt)
	lb_Value2[li_Cnt] = of_getbit(al_Value2, li_Cnt)
Next

For li_Cnt = 1 To 32
	If lb_Value1[li_Cnt] And lb_Value2[li_Cnt] Then
		ll_Result = ll_Result + (2^(li_Cnt - 1))
	End If
Next

Return ll_Result

end function

public function string uf_getdept ();long ll_rtn
string ls_dept,ls_Account
oleobject ole_RTXAPI
oleobject ole_KernalRoot
oleobject ole_App
oleobject ole_Helper

ole_RTXAPI=create oleobject
ll_rtn=ole_RTXAPI.ConnectToNewObject("RTXClient.RTXAPI")//("RTXCAPI.RTXCRoot")//("RTXClient.RTXAPI")
if ll_rtn<>0 then
	of_connecterrormsg(ll_rtn,"RTXClient.RTXAPI")
	destroy ole_RTXAPI
	return ""
end if
TRY
	
 	ole_App=create oleobject
	ole_App = ole_RTXAPI.GetObject("AppRoot")
	
	ole_Helper=create oleobject
	ole_Helper = ole_App.GetAppObject("RTXHelper")
	
	ole_KernalRoot=create oleobject
	ole_KernalRoot = ole_RTXAPI.GetObject("KernalRoot")
	ls_Account=ole_KernalRoot.Account
	
	ls_dept = ole_Helper.GetBuddyDept(ls_Account)
	 
	
CATCH (throwable rtxloge)
FINALLY 
END TRY

if isvalid(ole_App) then destroy ole_App
if isvalid(ole_Helper) then destroy ole_Helper
if isvalid(ole_KernalRoot) then destroy ole_KernalRoot
ole_RTXAPI.disconnectobject()
destroy ole_RTXAPI

return ls_dept

end function
